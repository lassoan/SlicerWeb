# CLI modules of Slicer, built into this application.
#
# A CLI module is a separate program in desktop Slicer, and a web page cannot start programs. The
# modules here are the same C++ sources, compiled as a library that is linked into SlicerWebCore,
# and vtkSlicerCLIModuleLogic runs one exactly as it runs a shared object module on the desktop: it
# writes the chosen nodes to files, calls the module's entry point with the command line, and reads
# back the files it wrote. Everything that uses a CLI module through that logic - Crop Volume asks
# for Resample Scalar/Vector/DWI Volume from its C++ - therefore works unchanged.
#
# One module at a time: each CLI module defines ModuleEntryPoint and XMLModuleDescription, so two of
# them cannot be linked into the same library without renaming those (which the compile definitions
# below could do).

set(_cli_dir ${Slicer_SOURCE_DIR}/Modules/CLI)
set(_module ResampleScalarVectorDWIVolume)
set(_clp_dir ${CMAKE_CURRENT_SOURCE_DIR}/${_module})

#-----------------------------------------------------------------------------
# The command line parser of the module is generated by GenerateCLP from the module's XML. That
# program runs on the machine doing the build and needs a host ITK, which this toolchain does not
# have, so the generated file is kept in the tree. It carries a copy of the XML it was made from:
# if that no longer matches the module being built, it has to be generated again.
#-----------------------------------------------------------------------------
file(READ ${_clp_dir}/${_module}CLP.h _clp_source)
string(REGEX MATCH "XMLModuleDescription\\[\\] =([^;]*);" _matched "${_clp_source}")
set(_carried "${CMAKE_MATCH_1}")
string(REPLACE "\\\"" "@QUOTE@" _carried "${_carried}")    # keep the quotes that are part of the XML
string(REPLACE "\"" "" _carried "${_carried}")          # drop the ones that delimit each line
string(REPLACE "@QUOTE@" "\"" _carried "${_carried}")
file(READ ${_cli_dir}/${_module}/${_module}.xml _xml)
foreach(_text _carried _xml)
  # Compared without the line breaks the generator wrote into the string, and without the ones the
  # module's own description text contains, which look no different once it is a C string.
  string(REPLACE "\\n" "" ${_text} "${${_text}}")
  string(REGEX REPLACE "[ \t\r\n]+" "" ${_text} "${${_text}}")
endforeach()
if(NOT "${_carried}" STREQUAL "${_xml}")
  message(FATAL_ERROR
    "${_clp_dir}/${_module}CLP.h was generated from a different ${_module}.xml than the one being "
    "built. Generate it again with GenerateCLP (see the comment at the top of the file).")
endif()

#-----------------------------------------------------------------------------
add_library(SlicerWebCLIModules STATIC
  ${_cli_dir}/${_module}/${_module}.cxx
  ${_cli_dir}/ResampleDTIVolume/dtiprocessFiles/deformationfieldio.cxx
  )
target_include_directories(SlicerWebCLIModules PRIVATE
  ${_clp_dir}                                  # the generated command line parser
  ${Slicer_SOURCE_DIR}/Resources               # NAMICLogo.h, which the parser includes
  ${_cli_dir}/ResampleDTIVolume                # headers shared with the DTI resampler
  ${SlicerExecutionModel_INCLUDE_DIRS}         # ModuleProcessInformation.h and tclap
  )
# What makes the module's main() the entry point the logic calls (as SEMMacroBuildCLI does when it
# builds a CLI module as a library).
target_compile_definitions(SlicerWebCLIModules PRIVATE main=ModuleEntryPoint)
target_link_libraries(SlicerWebCLIModules PRIVATE ${ITK_LIBRARIES})
set_target_properties(SlicerWebCLIModules PROPERTIES POSITION_INDEPENDENT_CODE ON)
