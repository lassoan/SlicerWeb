# CLI modules of Slicer, built into this application.
#
# A CLI module is a separate program in desktop Slicer, and a web page cannot start programs. The
# modules here are the same C++ sources, compiled into libraries that SlicerWebCore links, and
# vtkSlicerCLIModuleLogic runs one exactly as it runs a shared object module on the desktop: it
# writes the chosen nodes to files, calls the module's entry point with the command line, and reads
# back the files it wrote. Everything that uses a CLI module through that logic - Crop Volume asks
# for Resample Scalar/Vector/DWI Volume from its C++ - therefore works unchanged.
#
# Each CLI module declares the same handful of names - its entry point, the XML it describes itself
# with, its logo - because each is normally a library of its own. Here they share one, so while a
# module is compiled every one of those names is given the module's name as a prefix, and
# vtkSlicerWebCLIModule.cxx knows them by those names.
#
# The command line parser of a module is generated by GenerateCLP from its XML. That program runs on
# the machine doing the build and needs a host ITK, which this toolchain does not have, so the
# generated file is kept beside this one instead. It carries a copy of the XML it was made from,
# which is checked below against the module being built.

set(_cli_dir ${Slicer_SOURCE_DIR}/Modules/CLI)

function(slicerweb_add_cli_module name)
  cmake_parse_arguments(ARG "" "" "SOURCES;INCLUDE_DIRECTORIES" ${ARGN})
  set(_clp ${CMAKE_CURRENT_SOURCE_DIR}/${name}/${name}CLP.h)

  # Is the parser that is kept here still the one this module's XML describes?
  file(READ ${_clp} _clp_source)
  string(REGEX MATCH "XMLModuleDescription\\[\\] =([^;]*);" _matched "${_clp_source}")
  set(_carried "${CMAKE_MATCH_1}")
  string(REPLACE "\\\"" "@QUOTE@" _carried "${_carried}")  # keep the quotes that are part of the XML
  string(REPLACE "\"" "" _carried "${_carried}")           # drop the ones that delimit each line
  string(REPLACE "@QUOTE@" "\"" _carried "${_carried}")
  file(READ ${_cli_dir}/${name}/${name}.xml _xml)
  foreach(_text _carried _xml)
    # Compared without the line breaks the generator wrote into the string, and without the ones the
    # module's own description text contains, which look no different once it is a C string.
    string(REPLACE "\\n" "" ${_text} "${${_text}}")
    string(REGEX REPLACE "[ \t\r\n]+" "" ${_text} "${${_text}}")
  endforeach()
  if(NOT "${_carried}" STREQUAL "${_xml}")
    message(FATAL_ERROR "${_clp} was generated from a different ${name}.xml than the one being built. "
                        "Generate it again with GenerateCLP (see the comment at the top of it).")
  endif()

  add_library(SlicerWebCLI${name} STATIC ${_cli_dir}/${name}/${name}.cxx ${ARG_SOURCES})
  target_include_directories(SlicerWebCLI${name} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/${name}           # the generated command line parser
    ${Slicer_SOURCE_DIR}/Resources                # the logo it includes
    ${SlicerExecutionModel_INCLUDE_DIRS}          # ModuleProcessInformation.h and tclap
    ${Slicer_SOURCE_DIR}/Base/CLI                 # itkPluginUtilities.h, shared by the CLI modules
    ${ARG_INCLUDE_DIRECTORIES}
    )
  # main=ModuleEntryPoint makes the module's main() the entry point the logic calls, as
  # SEMMacroBuildCLI does when it builds a CLI module as a library. The rest keep two modules in one
  # library from defining each other's names.
  set(_definitions main=ModuleEntryPoint)
  foreach(_symbol ModuleEntryPoint XMLModuleDescription GetXMLModuleDescription GetModuleLogo
                  ModuleLogoWidth ModuleLogoHeight ModuleLogoPixelSize ModuleLogoLength ModuleLogoImage
                  splitString splitFilenames)
    list(APPEND _definitions ${_symbol}=${name}_${_symbol})
  endforeach()
  target_compile_definitions(SlicerWebCLI${name} PRIVATE ${_definitions})
  target_link_libraries(SlicerWebCLI${name} PRIVATE ${ITK_LIBRARIES})
  set_target_properties(SlicerWebCLI${name} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endfunction()

slicerweb_add_cli_module(ResampleScalarVectorDWIVolume
  SOURCES ${_cli_dir}/ResampleDTIVolume/dtiprocessFiles/deformationfieldio.cxx
  INCLUDE_DIRECTORIES ${_cli_dir}/ResampleDTIVolume   # headers shared with the DTI resampler
  )
slicerweb_add_cli_module(MedianImageFilter)

# The libraries SlicerWebCore links; vtkSlicerWebCLIModule.cxx lists the same modules.
set(SlicerWeb_CLI_MODULE_LIBRARIES
  SlicerWebCLIResampleScalarVectorDWIVolume
  SlicerWebCLIMedianImageFilter
  CACHE INTERNAL "" FORCE)
